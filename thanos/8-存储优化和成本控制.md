```yaml

  # 热数据：频繁访问，使用高性能存储
```

#### 压缩调度策略

```bash
#!/bin/bash
# compaction-scheduler.sh - 智能压缩调度

# 根据负载情况调整压缩策略
CURRENT_LOAD=$(uptime | awk -F'load average:' '{print $2}' | cut -d, -f1 | tr -d ' ')

if (( $(echo "$CURRENT_LOAD < 1.0" | bc -l) )); then
    # 低负载时执行全量压缩
    echo "执行全量压缩..."
    thanos compact \\
        --objstore.config-file=./s3-config.yaml \\
        --compact.concurrency=8 \\
        --wait
        
elif (( $(echo "$CURRENT_LOAD < 3.0" | bc -l) )); then
    # 中等负载时执行增量压缩
    echo "执行增量压缩..."
    thanos compact \\
        --objstore.config-file=./s3-config.yaml \\
        --compact.concurrency=4 \\
        --compact.delay=1h
        
else
    # 高负载时暂停压缩
    echo "系统负载过高，暂停压缩操作"
fi
```

```bash
#!/bin/bash
# cost-optimizer.sh - 存储成本优化器

# 分析存储使用情况
analyze_storage_usage() {
    echo "分析存储使用情况..."
    
    # 获取存储桶大小
    BUCKET_SIZE=$(aws s3 ls --summarize --human-readable --recursive s3://thanos-bucket | grep "Total Size" | awk '{print $3}')
    
    # 计算存储成本
    COST=$(echo "$BUCKET_SIZE * 0.023" | bc)
    
    echo "当前存储大小: $BUCKET_SIZE"
    echo "预估月成本: $COST 美元"
    
    # 检查优化机会
    if (( $(echo "$COST > 500" | bc -l) )); then
        echo "检测到成本优化机会"
        optimize_storage
    fi
}

# 执行存储优化
optimize_storage() {
    echo "执行存储优化..."
    
    # 迁移冷数据到低频存储
    echo "迁移冷数据到STANDARD_IA..."
    aws s3 cp --storage-class STANDARD_IA --recursive s3://thanos-bucket/prometheus/ s3://thanos-bucket/prometheus-ia/
    
    # 删除过期数据
    echo "删除过期数据..."
    find_expired_data | xargs -I {} aws s3 rm s3://thanos-bucket/{}
    
    # 压缩历史数据
    echo "压缩历史数据..."
    thanos compact --objstore.config-file=./s3-config.yaml --compact.concurrency=2
}

# 主函数
main() {
    analyze_storage_usage
}

main "$@"
```

## 8.5 容量规划和预测

### 8.5.1 容量预测模型

#### 基于历史数据的预测

```python
# capacity-predictor.py - 容量预测工具
import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

def predict_capacity():
    # 加载历史数据
    df = pd.read_csv('storage_usage.csv')
    
    # 准备特征和目标变量
    X = df[['days', 'metrics_count', 'samples_per_second']]
    y = df['storage_gb']
    
    # 训练预测模型
    model = LinearRegression()
    model.fit(X, y)
    
    # 预测未来容量
    future_days = 365
    future_metrics = df['metrics_count'].iloc[-1] * 1.1  # 10%增长
    future_samples = df['samples_per_second'].iloc[-1] * 1.05  # 5%增长
    
    future_X = [[future_days, future_metrics, future_samples]]
    predicted_capacity = model.predict(future_X)[0]
    
    return predicted_capacity

# 计算成本预测
def predict_cost(storage_gb, days=30):
    # 存储成本
    storage_cost = storage_gb * 0.023  # 标准存储价格
    
    # 请求成本（估算）
    requests_per_day = 10000  # 日均请求数
    request_cost = requests_per_day * days * 0.0004
    
    # 总成本
    total_cost = storage_cost + request_cost
    
    return total_cost

if __name__ == "__main__":
    capacity = predict_capacity()
    cost = predict_cost(capacity)
    
    print(f"预测一年后存储需求: {capacity:.2f} GB")
    print(f"预估月成本: ${cost:.2f}")
```
