```
├─────────────────┤    ├─────────────────┤    ├─────────────────┤
│   Sidecar A1    │    │   Sidecar B1    │    │   Sidecar C1    │
│   Sidecar A2    │    │   Sidecar B2    │    │   Sidecar C2    │
```

```yaml
# query-ha-deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: thanos-query
  namespace: monitoring
spec:
  replicas: 3
  selector:
    matchLabels:
      app: thanos-query
  template:
    metadata:
      labels:
        app: thanos-query
    spec:
      containers:
      - name: thanos-query
        image: thanosio/thanos:latest
        args:
        - query
        - --http-address=0.0.0.0:19192
        - --grpc-address=0.0.0.0:19091
        - --query.replica-label=replica
        - --store=dnssrv+_grpc._tcp.thanos-sidecar.monitoring.svc.cluster.local
        - --store=dnssrv+_grpc._tcp.thanos-store.monitoring.svc.cluster.local
        - --query.auto-downsampling
        - --query.partial-response
        - --log.level=info
        ports:
        - containerPort: 19192
          name: http
        - containerPort: 19091
          name: grpc
        livenessProbe:
          httpGet:
            path: /-/healthy
            port: http
          initialDelaySeconds: 30
          periodSeconds: 10
        readinessProbe:
          httpGet:
            path: /-/ready
            port: http
          initialDelaySeconds: 5
          periodSeconds: 5
---

kind: Service
metadata:
  name: thanos-query
  namespace: monitoring
  annotations:
    service.beta.kubernetes.io/aws-load-balancer-type: "nlb"

  selector:
    app: thanos-query
  ports:
  - name: http
    port: 19192
    targetPort: http
    protocol: TCP
  - name: grpc
    port: 19091
```

## 6.3 负载均衡和故障转移

### 6.3.1 负载均衡策略

**多层级负载均衡**：
```yaml
# 应用层负载均衡（Nginx）
upstream thanos_query {
    # 基于响应时间的负载均衡
    least_conn;
    
    server thanos-query-1:19192 max_fails=3 fail_timeout=30s;
    server thanos-query-2:19192 max_fails=3 fail_timeout=30s;
    server thanos-query-3:19192 max_fails=3 fail_timeout=30s;
    
    # 健康检查
    check interval=3000 rise=2 fall=3 timeout=1000 type=http;
    check_http_send "HEAD /-/healthy HTTP/1.0\r\n\r\n";
    check_http_expect_alive http_2xx http_3xx;
}

server {
    listen 80;
    server_name thanos.example.com;
    
    location / {
        proxy_pass http://thanos_query;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        
        # 超时配置
        proxy_connect_timeout 30s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
        
        # 错误处理
        proxy_next_upstream error timeout invalid_header http_500 http_502 http_503 http_504;
        proxy_next_upstream_tries 3;
        proxy_next_upstream_timeout 30s;
    }
}
```

## 6.4 监控和告警配置

### 6.4.1 集群健康监控

**关键集群指标**：
```yaml
groups:
- name: thanos-cluster-health
  rules:
  - record: thanos_cluster_health_score
    expr: |
      (
        # Query组件健康度
        (avg(up{job="thanos-query"}) * 0.3) +
        
        # Store组件健康度  
        (avg(up{job="thanos-store"}) * 0.3) +
        
        # Sidecar连接健康度
        (avg(thanos_query_store_apis_dns_provider_results{result="success"}) * 0.2) +
        
        # 对象存储健康度
        (avg(thanos_objstore_bucket_operations_total{operation="get",le="0.1"}) * 0.2)
      ) * 100
    
  - record: thanos_query_load_distribution
    expr: |
      sum by (instance) (rate(thanos_query_requests_total[5m])) /
      sum(rate(thanos_query_requests_total[5m]))
    
  - record: thanos_store_shard_utilization
    expr: |
      thanos_store_bucket_store_series_loaded / 
      thanos_store_bucket_store_series_max
```

### 6.4.2 高可用告警规则

**集群故障告警**：
```yaml
groups:
- name: thanos-ha-alerts
  rules:
  - alert: ThanosQueryInstanceDown
    expr: up{job="thanos-query"} == 0
    for: 2m
    labels:
      severity: critical
    annotations:
      summary: "Thanos Query实例宕机"
      description: "{{ $labels.instance }} Query实例已宕机2分钟"
      
  - alert: ThanosStoreUnhealthy
    expr: thanos_store_bucket_store_series_loaded / thanos_store_bucket_store_series_max > 0.9
    for: 5m
    labels:
      severity: warning
    annotations:
      summary: "Thanos Store分片负载过高"
      description: "Store分片 {{ $labels.instance }} 负载超过90%"
      
  - alert: ThanosClusterUnbalanced
    expr: stddev(thanos_query_load_distribution) > 0.3
    for: 10m
    labels:
      severity: warning
    annotations:
      summary: "Thanos集群负载不均衡"
      description: "Query实例间负载分布标准差超过30%"
      
  - alert: ThanosStorageLatencyHigh
    expr: histogram_quantile(0.95, thanos_objstore_bucket_operation_duration_seconds_bucket{operation="get"}) > 5
    for: 5m
    labels:
      severity: critical
    annotations:
      summary: "Thanos存储延迟过高"
      description: "对象存储get操作P95延迟超过5秒"
```

### 6.6.2 容量规划指南

**容量规划公式**：
```
# 存储容量计算
总存储需求 = 原始数据量 × 保留时间 × 压缩比 × 冗余因子

# 示例计算
原始数据量: 100GB/天
保留时间: 30天
压缩比: 0.3 (70%压缩)
冗余因子: 1.5 (50%冗余)

总存储需求 = 100 × 30 × 0.3 × 1.5 = 1.35TB

# 内存需求计算
Query内存 = 并发查询数 × 平均查询内存 × 安全系数
Store内存 = 索引大小 + 缓存大小 + 工作内存

# 网络带宽计算
所需带宽 = 数据摄入速率 + 查询流量 + 复制流量
```
